   549→// Very small heuristic "AI" response generator (client-side, deterministic-ish).
   550→function generateAssistantReply(userText: string): { content: string; suggestedTokens?: string[] } {
   551→  const lower = userText.toLowerCase();
   552→  // Try to find a token mentioned in the user message.
   553→  const matched = TOKENS.find(
   554→    (t) =>
   555→      lower.includes(`$${t.symbol.toLowerCase()}`) ||
   556→      lower.includes(t.symbol.toLowerCase()) ||
   557→      lower.includes(t.name.toLowerCase())
   558→  );
   559→
   560→  if (matched) {
   561→    const bull = matched.change24h >= 0;
   562→    const inflowAbs = Math.abs(matched.smartMoneyInflow24h);
   563→    const inflowStr =
   564→      inflowAbs >= 1_000_000
   565→        ? `$${(inflowAbs / 1_000_000).toFixed(2)}M`
   566→        : `$${(inflowAbs / 1000).toFixed(0)}K`;
   567→    const smart =
   568→      matched.smartMoneyInflow24h >= 0
   569→        ? `Smart money has net ${bull ? "accumulated" : "distributed"} ${inflowStr} over the last 24h across ${matched.smartMoneyHolders} tracked wallets.`
   570→        : `Smart money has net distributed ${inflowStr} over the last 24h.`;
   571→    return {
   572→      content: `${matched.name} (${matched.symbol}) is trading at $${matched.price.toFixed(
   573→        matched.price < 1 ? 4 : 2
   574→      )}, ${bull ? "up" : "down"} ${Math.abs(matched.change24h).toFixed(2)}% in 24h. ${smart} ${
   575→        matched.smartMoneyHolders >= 100
   576→          ? "That's a healthy concentration of smart money."
   577→          : "Smart money concentration is low — proceed with caution."
   578→      } Liquidity: $${(matched.liquidity / 1_000_000).toFixed(1)}M. Volume 24h: $${(
   579→          matched.volume24h / 1_000_000
   580→        ).toFixed(0)}M.`,
   581→      suggestedTokens: [matched.id],
   582→    };
   583→  }
