   540→// Very small heuristic "AI" response generator (client-side, deterministic-ish).
   541→function generateAssistantReply(userText: string): { content: string; suggestedTokens?: string[] } {
   542→  const lower = userText.toLowerCase();
   543→  // Try to find a token mentioned in the user message.
   544→  const matched = TOKENS.find(
   545→    (t) =>
   546→      lower.includes(`$${t.symbol.toLowerCase()}`) ||
   547→      lower.includes(t.symbol.toLowerCase()) ||
   548→      lower.includes(t.name.toLowerCase())
   549→  );
   550→
   551→  if (matched) {
   552→    const bull = matched.change24h >= 0;
   553→    const inflowAbs = Math.abs(matched.smartMoneyInflow24h);
   554→    const inflowStr =
   555→      inflowAbs >= 1_000_000
   556→        ? `$${(inflowAbs / 1_000_000).toFixed(2)}M`
   557→        : `$${(inflowAbs / 1000).toFixed(0)}K`;
   558→    const smart =
   559→      matched.smartMoneyInflow24h >= 0
   560→        ? `Smart money has net ${bull ? "accumulated" : "distributed"} ${inflowStr} over the last 24h across ${matched.smartMoneyHolders} tracked wallets.`
   561→        : `Smart money has net distributed ${inflowStr} over the last 24h.`;
   562→    return {
   563→      content: `${matched.name} (${matched.symbol}) is trading at $${matched.price.toFixed(
   564→        matched.price < 1 ? 4 : 2
   565→      )}, ${bull ? "up" : "down"} ${Math.abs(matched.change24h).toFixed(2)}% in 24h. ${smart} ${
   566→        matched.smartMoneyHolders >= 100
   567→          ? "That's a healthy concentration of smart money."
   568→          : "Smart money concentration is low — proceed with caution."
   569→      } Liquidity: $${(matched.liquidity / 1_000_000).toFixed(1)}M. Volume 24h: $${(
   570→          matched.volume24h / 1_000_000
   571→        ).toFixed(0)}M.`,
   572→      suggestedTokens: [matched.id],
   573→    };
   574→  }
   575→
   576→  if (lower.includes("portfolio")) {
   577→    return {
   578→      content:
   579→        "Your portfolio is up 4.2% in the last 24h, led by WIF (+14.3%) and BONK (+22.1%). Your largest concentration is SOL at 32% — consider trimming if you want to reduce L1 beta. You have 3 untracked smart-money alerts waiting in your Signals tab.",
   580→    };
   581→  }
