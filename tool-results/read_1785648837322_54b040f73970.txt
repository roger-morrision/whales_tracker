   171→// Default portfolio holdings seeded from the static PORTFOLIO cryptoHoldings.
   172→// These get mutated by trades; if no trades have happened, this matches the
   173→// previous static display exactly.
   174→function initialPortfolioHoldings(): PortfolioHolding[] {
   175→  return (PORTFOLIO.cryptoHoldings || []).map((h) => ({
   176→    tokenId: h.tokenId,
   177→    amount: h.amount,
   178→    costUsd: h.amount * (h.avgCost ?? 0),
   179→  }));
   180→}
